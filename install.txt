##############################
# Linux
# The below example is for 'apt' based distros.
# Other package managers on other distros will be similar.
##############################
sudo apt update

sudo apt install python3-full python3-venv python3-pip git

## 2. Clone the repository
cd /opt # or where you want to install it.

git clone https://github.com/ChrisMcGowanAu/PyTkQuickGui.git

cd /opt/PyTkQuickGui ( or the directory it was installed in)

## 3. Install required packages
python3 -m venv venv
venv/bin/pip install --upgrade pip
venv/bin/pip install -r requirments.txt

Note: ttkbootstrap has to be version 2.0 or later

In some environments, running the test python file or the generated python man not work, If you hit this issue, do the following:

Summary: build a .venv environment in your home directory 
and add it to your $PATH

# In your home directory run the following:
python3 -m venv .venv
.venv/bin/pip install --upgrade pip

# Assuming you installed the tool in  /opt
.venv/bin/pip install -r /opt/PyTkQuickGui/requirments.txt  

# Modify your .bashrc, or equivelant file for the shell you use
# And add this line. 
export PATH=$HOME/.venv/bin/:$PATH

logout and login again, have a look at $PATH to see if it is correct
echo $PATH
